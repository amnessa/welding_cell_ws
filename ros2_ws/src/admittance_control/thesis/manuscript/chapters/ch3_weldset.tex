\section{Seams by the Accessibility Rule}
\label{sec:ws_d4}
For two planar faces with unit normals $\vect{n}_1,\vect{n}_2$ and points
$\vect{p}_1,\vect{p}_2$, the nominal seam is the line of intersection of their planes:
\begin{equation}
\vect{d} = \frac{\vect{n}_1\times\vect{n}_2}{\norm{\vect{n}_1\times\vect{n}_2}},\qquad
\vect{x}(s) = \vect{x}_0 + s\,\vect{d},\quad
\vect{n}_i^{\top}\vect{x}_0 = \vect{n}_i^{\top}\vect{p}_i\ (i=1,2).
\label{eq:plane_intersection}
\end{equation}

\section{Tack Placement Rule}
\label{sec:ws_tacks}
With $t$ the thinner member and $L$ the seam length (open seam), tackrule-0.1:
\begin{gather}
\ell_{\mathrm{tack}} = \operatorname{clip}(4t,\,10,\,50)\ \mathrm{mm},\qquad
d_{\max} = \min(33t,\,400)\ \mathrm{mm},\qquad d_{\min} = 10t,
\label{eq:tackrule_len}\\
m = \max(2t,\ \ell_{\mathrm{tack}}),\qquad
n = \max\!\left(2,\ \ceil*{\frac{L-2m}{d_{\max}}}+1\right).
\label{eq:tackrule_n}
\end{gather}
